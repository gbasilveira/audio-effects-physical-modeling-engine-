cmake_minimum_required(VERSION 3.20)
project(AudioEffectsPhysicalModelingEngine VERSION 0.1.0 LANGUAGES CXX)
option(AUDIOFX_SANITIZE "Enable address/undefined behaviour sanitizers" OFF)
add_library(audiofx src/engine.cpp)
target_include_directories(audiofx PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(audiofx PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(audiofx PRIVATE /W4)
else()
  target_compile_options(audiofx PRIVATE -Wall -Wextra -Wpedantic)
  if(AUDIOFX_SANITIZE)
    target_compile_options(audiofx PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(audiofx PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(audiofx-render tools/render.cpp)
target_link_libraries(audiofx-render PRIVATE audiofx)
include(CTest)
if(BUILD_TESTING)
  add_executable(audiofx-tests tests/test_engine.cpp)
  target_link_libraries(audiofx-tests PRIVATE audiofx)
  add_test(NAME audiofx-core COMMAND audiofx-tests)
endif()
install(TARGETS audiofx audiofx-render EXPORT audiofxTargets)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT audiofxTargets NAMESPACE audiofx:: DESTINATION lib/cmake/audiofx)
add_executable(audiofx-benchmark tools/benchmark.cpp)
target_link_libraries(audiofx-benchmark PRIVATE audiofx)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/audiofxConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/audiofxConfig.cmake INSTALL_DESTINATION lib/cmake/audiofx)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/audiofxConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/audiofxConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/audiofxConfigVersion.cmake DESTINATION lib/cmake/audiofx)
